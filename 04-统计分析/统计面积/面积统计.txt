    def open_tiff_file(self):
        """打开单个TIFF文件"""
        file_path = filedialog.askopenfilename(
            title="选择TIFF文件",
            filetypes=[("TIFF文件", "*.tif *.tiff"), ("所有文件", "*.*")]
        )

        if file_path:
            self.load_tiff_file(file_path)

    def open_folder(self):
        """打开文件夹选择多个TIFF文件"""
        folder_path = filedialog.askdirectory(title="选择包含TIFF文件的文件夹")

        if folder_path:
            # 查找文件夹中的所有TIFF文件
            tiff_files = []
            for ext in ['.tif', '.tiff', '.TIF', '.TIFF']:
                tiff_files.extend(Path(folder_path).glob(f'*{ext}'))

            if not tiff_files:
                messagebox.showwarning("警告", "文件夹中没有找到TIFF文件")
                return

            # 按文件名排序
            tiff_files = sorted(tiff_files)

            # 询问是否合并为堆叠
            if messagebox.askyesno("加载方式", f"找到 {len(tiff_files)} 个TIFF文件。\n\n是否将它们合并为一个堆叠？"):
                self.load_tiff_stack_from_files(tiff_files)
            else:
                # 只加载第一个文件
                self.load_tiff_file(str(tiff_files[0]))

    def load_tiff_file(self, file_path):
        """加载TIFF文件"""
        try:
            self.status_var.set("正在加载文件...")
            self.root.update()

            self.file_path = file_path

            # 使用tifffile读取TIFF
            with tifffile.TiffFile(file_path) as tif:
                self.tiff_stack = tif.asarray()

                # 获取图像形状
                if len(self.tiff_stack.shape) == 2:
                    # 单帧图像
                    self.tiff_stack = self.tiff_stack[np.newaxis, ...]
                    self.total_frames = 1
                elif len(self.tiff_stack.shape) == 3:
                    # 堆叠图像 (frames, height, width) 或 (height, width, frames)
                    if self.tiff_stack.shape[2] < self.tiff_stack.shape[0] and self.tiff_stack.shape[2] < \
                            self.tiff_stack.shape[1]:
                        # 可能是 (height, width, frames) 格式
                        self.tiff_stack = np.moveaxis(self.tiff_stack, -1, 0)

                    self.total_frames = self.tiff_stack.shape[0]
                elif len(self.tiff_stack.shape) == 4:
                    # 彩色堆叠图像 (frames, height, width, channels)
                    self.total_frames = self.tiff_stack.shape[0]
                else:
                    messagebox.showerror("错误", f"不支持的图像维度: {self.tiff_stack.shape}")
                    return

            # 重置状态
            self.current_frame = 0
            self.polygons = {}
            self.polygon_colors = {}
            self.measurements = {}
            self.next_polygon_id = 1
            self.drawing_mode = False
            self.current_polygon_points = []
            self.zoom_factor = 1.0
            self.pan_offset = [0, 0]

            # 更新UI
            self.frame_spin.config(from_=1, to=self.total_frames)
            self.frame_var.set("1")

            # 更新文件信息
            self.update_file_info()

            # 显示第一帧
            self.display_current_frame()

            self.status_var.set(f"已加载: {Path(file_path).name}")

        except Exception as e:
            messagebox.showerror("错误", f"加载文件失败: {str(e)}")
            self.status_var.set("加载失败")

    def load_tiff_stack_from_files(self, file_paths):
        """从多个文件加载堆叠"""
        try:
            self.status_var.set("正在加载堆叠...")
            self.root.update()

            frames = []
            for i, file_path in enumerate(file_paths):
                with tifffile.TiffFile(file_path) as tif:
                    img = tif.asarray()
                    if len(img.shape) == 2:
                        frames.append(img)
                    elif len(img.shape) == 3:
                        # 如果是彩色图像，转换为灰度
                        if img.shape[2] == 3:
                            # RGB转灰度
                            gray = np.dot(img[..., :3], [0.2989, 0.5870, 0.1140])
                            frames.append(gray)
                        else:
                            frames.append(img[..., 0])
                    else:
                        messagebox.showwarning("警告", f"跳过不支持的图像: {file_path}")

                # 更新进度
                if i % 10 == 0:
                    self.status_var.set(f"正在加载: {i + 1}/{len(file_paths)}")
                    self.root.update()

            if not frames:
                messagebox.showerror("错误", "没有有效的图像文件")
                return

            # 堆叠所有帧
            self.tiff_stack = np.stack(frames)
            self.total_frames = len(frames)
            self.file_path = str(file_paths[0].parent)

            # 重置状态
            self.current_frame = 0
            self.polygons = {}
            self.polygon_colors = {}
            self.measurements = {}
            self.next_polygon_id = 1
            self.drawing_mode = False
            self.current_polygon_points = []
            self.zoom_factor = 1.0
            self.pan_offset = [0, 0]

            # 更新UI
            self.frame_spin.config(from_=1, to=self.total_frames)
            self.frame_var.set("1")

            # 更新文件信息
            self.update_file_info()

            # 显示第一帧
            self.display_current_frame()

            self.status_var.set(f"已加载堆叠: {len(frames)}帧")

        except Exception as e:
            messagebox.showerror("错误", f"加载堆叠失败: {str(e)}")
            self.status_var.set("加载失败")

    def update_file_info(self):
        """更新文件信息显示"""
        self.info_text.config(state=tk.NORMAL)
        self.info_text.delete(1.0, tk.END)

        if self.tiff_stack is not None:
            info = f"文件: {Path(self.file_path).name}\n"
            info += f"总帧数: {self.total_frames}\n"
            info += f"图像尺寸: {self.tiff_stack.shape[2]} x {self.tiff_stack.shape[1]}\n"
            info += f"数据类型: {self.tiff_stack.dtype}\n"

            if len(self.tiff_stack.shape) == 4:
                info += f"通道数: {self.tiff_stack.shape[3]}\n"

            info += f"值范围: {self.tiff_stack.min():.2f} - {self.tiff_stack.max():.2f}"

            self.info_text.insert(tk.END, info)

        self.info_text.config(state=tk.DISABLED)

    def display_current_frame(self):
        """显示当前帧"""
        if self.tiff_stack is None:
            return

        try:
            # 清除当前图像
            self.ax.clear()

            # 获取当前帧
            frame_idx = self.current_frame
            img_data = self.tiff_stack[frame_idx]

            # 处理不同维度的图像
            if len(img_data.shape) == 2:
                # 灰度图像
                display_img = img_data
                cmap = 'gray'
            elif len(img_data.shape) == 3:
                # 彩色图像
                if img_data.shape[2] == 3:
                    # RGB图像
                    display_img = img_data
                    cmap = None
                else:
                    # 多通道图像，显示第一个通道
                    display_img = img_data[:, :, 0]
                    cmap = 'gray'
            else:
                display_img = img_data
                cmap = 'gray'

            # 应用缩放和平移
            extent = [
                -self.pan_offset[0] / self.zoom_factor,
                (display_img.shape[1] - self.pan_offset[0]) / self.zoom_factor,
                (display_img.shape[0] - self.pan_offset[1]) / self.zoom_factor,
                -self.pan_offset[1] / self.zoom_factor
            ]

            # 显示图像
            self.ax.imshow(display_img, cmap=cmap, extent=extent, aspect='auto')

            # 设置标题
            self.ax.set_title(f"帧 {frame_idx + 1}/{self.total_frames}")

            # 隐藏坐标轴
            self.ax.set_xticks([])
            self.ax.set_yticks([])

            # 绘制当前帧的多边形
            if frame_idx in self.polygons:
                for poly_id, points in self.polygons[frame_idx].items():
                    if points and len(points) >= 3:
                        # 转换坐标为显示坐标
                        display_points = []
                        for x, y in points:
                            display_x = (x - self.pan_offset[0]) / self.zoom_factor
                            display_y = (display_img.shape[0] - y - self.pan_offset[1]) / self.zoom_factor
                            display_points.append([display_x, display_y])

                        # 创建多边形
                        polygon = MplPolygon(display_points,
                                             closed=True,
                                             edgecolor=self.polygon_colors.get(poly_id, 'red'),
                                             facecolor=self.polygon_colors.get(poly_id, 'red') + '20',  # 添加透明度
                                             linewidth=2,
                                             alpha=0.5)
                        self.ax.add_patch(polygon)

                        # 添加标签
                        if points:
                            centroid_x = sum(p[0] for p in points) / len(points)
                            centroid_y = sum(p[1] for p in points) / len(points)
                            display_centroid_x = (centroid_x - self.pan_offset[0]) / self.zoom_factor
                            display_centroid_y = (display_img.shape[0] - centroid_y - self.pan_offset[
                                1]) / self.zoom_factor

                            self.ax.text(display_centroid_x, display_centroid_y,
                                         str(poly_id),
                                         color='white',
                                         fontsize=10, fontweight='bold',
                                         ha='center', va='center',
                                         bbox=dict(boxstyle="round,pad=0.3",
                                                   facecolor=self.polygon_colors.get(poly_id, 'red'),
                                                   alpha=0.8))

            # 绘制当前正在绘制的多边形
            if self.drawing_mode and self.current_polygon_points:
                points = self.current_polygon_points
                if len(points) >= 2:
                    # 绘制线段
                    for i in range(len(points) - 1):
                        x1, y1 = points[i]
                        x2, y2 = points[i + 1]
                        display_x1 = (x1 - self.pan_offset[0]) / self.zoom_factor
                        display_y1 = (display_img.shape[0] - y1 - self.pan_offset[1]) / self.zoom_factor
                        display_x2 = (x2 - self.pan_offset[0]) / self.zoom_factor
                        display_y2 = (display_img.shape[0] - y2 - self.pan_offset[1]) / self.zoom_factor

                        self.ax.plot([display_x1, display_x2], [display_y1, display_y2],
                                     'r-', linewidth=2, alpha=0.8)

                    # 绘制点
                    for x, y in points:
                        display_x = (x - self.pan_offset[0]) / self.zoom_factor
                        display_y = (display_img.shape[0] - y - self.pan_offset[1]) / self.zoom_factor
                        self.ax.plot(display_x, display_y, 'ro', markersize=6, alpha=0.8)

            # 更新画布
            self.canvas.draw()

            # 更新多边形列表
            self.update_polygon_list()

            # 更新统计信息
            self.update_statistics()

        except Exception as e:
            print(f"显示图像时出错: {str(e)}")

    def on_frame_changed(self):
        """帧数改变时调用"""
        try:
            frame_num = int(self.frame_var.get()) - 1
            if 0 <= frame_num < self.total_frames:
                self.current_frame = frame_num
                self.display_current_frame()
        except:
            pass

    def prev_frame(self):
        """上一帧"""
        if self.tiff_stack is not None and self.current_frame > 0:
            self.current_frame -= 1
            self.frame_var.set(str(self.current_frame + 1))
            self.display_current_frame()

    def next_frame(self):
        """下一帧"""
        if self.tiff_stack is not None and self.current_frame < self.total_frames - 1:
            self.current_frame += 1
            self.frame_var.set(str(self.current_frame + 1))
            self.display_current_frame()

    def toggle_drawing_mode(self):
        """切换绘制模式"""
        self.drawing_mode = not self.drawing_mode
        if self.drawing_mode:
            self.draw_btn.config(text="取消绘制")
            self.current_polygon_points = []
            self.status_var.set("绘制模式: 点击图像添加多边形顶点")
        else:
            self.draw_btn.config(text="绘制多边形")
            self.current_polygon_points = []
            self.status_var.set("就绪")

        self.display_current_frame()

    def on_canvas_click(self, event):
        """画布点击事件"""
        if not self.drawing_mode or self.tiff_stack is None:
            return

        if event.inaxes != self.ax:
            return

        # 获取图像坐标
        img_height = self.tiff_stack.shape[1]
        x = event.xdata * self.zoom_factor + self.pan_offset[0]
        y = img_height - (event.ydata * self.zoom_factor + self.pan_offset[1])

        # 添加点到当前多边形
        self.current_polygon_points.append((x, y))

        # 重新显示
        self.display_current_frame()

    def on_canvas_motion(self, event):
        """画布鼠标移动事件"""
        if self.tiff_stack is None:
            return

        if event.inaxes == self.ax:
            # 获取图像坐标
            img_height = self.tiff_stack.shape[1]
            x = event.xdata * self.zoom_factor + self.pan_offset[0] if event.xdata else 0
            y = img_height - (event.ydata * self.zoom_factor + self.pan_offset[1]) if event.ydata else 0

            self.coord_var.set(f"坐标: ({int(x)}, {int(y)})")
        else:
            self.coord_var.set("坐标: (0, 0)")

    def finish_polygon(self):
        """完成多边形绘制"""
        if not self.drawing_mode or len(self.current_polygon_points) < 3:
            messagebox.showwarning("警告", "至少需要3个点来创建多边形")
            return

        frame_idx = self.current_frame

        # 初始化帧的多边形字典
        if frame_idx not in self.polygons:
            self.polygons[frame_idx] = {}

        # 分配多边形ID
        poly_id = self.next_polygon_id
        self.next_polygon_id += 1

        # 保存多边形
        self.polygons[frame_idx][poly_id] = self.current_polygon_points.copy()

        # 分配颜色
        color_idx = (poly_id - 1) % len(self.color_palette)
        self.polygon_colors[poly_id] = self.color_palette[color_idx]

        # 计算面积
        area = self.calculate_polygon_area(self.current_polygon_points)

        # 保存测量结果
        if frame_idx not in self.measurements:
            self.measurements[frame_idx] = {}
        self.measurements[frame_idx][poly_id] = area

        # 重置绘制状态
        self.current_polygon_points = []
        self.drawing_mode = False
        self.draw_btn.config(text="绘制多边形")

        # 更新显示
        self.display_current_frame()

        self.status_var.set(f"已创建多边形 #{poly_id}, 面积: {area:.2f} 像素²")

    def calculate_polygon_area(self, points):
        """计算多边形面积（使用鞋带公式）"""
        if len(points) < 3:
            return 0

        area = 0
        n = len(points)

        for i in range(n):
            x1, y1 = points[i]
            x2, y2 = points[(i + 1) % n]
            area += x1 * y2 - x2 * y1

        return abs(area) / 2

    def update_polygon_list(self):
        """更新多边形列表"""
        self.poly_listbox.delete(0, tk.END)

        frame_idx = self.current_frame
        if frame_idx in self.polygons and self.polygons[frame_idx]:
            for poly_id in sorted(self.polygons[frame_idx].keys()):
                points = self.polygons[frame_idx][poly_id]
                area = self.measurements.get(frame_idx, {}).get(poly_id, 0)

                item_text = f"{poly_id:3d}  {area:8.2f}  {len(points):3d}"
                self.poly_listbox.insert(tk.END, item_text)

    def update_statistics(self):
        """更新统计信息"""
        self.stats_text.config(state=tk.NORMAL)
        self.stats_text.delete(1.0, tk.END)

        frame_idx = self.current_frame

        if frame_idx in self.measurements and self.measurements[frame_idx]:
            areas = list(self.measurements[frame_idx].values())

            stats = f"当前帧 #{frame_idx + 1}\n"
            stats += f"多边形数量: {len(areas)}\n"
            stats += f"总面积: {sum(areas):.2f} 像素²\n"
            stats += f"平均面积: {np.mean(areas):.2f} 像素²\n"
            stats += f"最小面积: {min(areas):.2f} 像素²\n"
            stats += f"最大面积: {max(areas):.2f} 像素²\n"
            stats += f"标准差: {np.std(areas):.2f} 像素²"
        else:
            stats = "当前帧无测量数据"

        self.stats_text.insert(tk.END, stats)
        self.stats_text.config(state=tk.DISABLED)

    def clear_current_frame(self):

    def delete_selected_polygon(self):
        """删除选中的多边形"""
        selection = self.poly_listbox.curselection()
        if not selection:
            messagebox.showwarning("警告", "请先选择一个多边形")
            return

        # 获取选中的多边形ID
        item_text = self.poly_listbox.get(selection[0])
        poly_id = int(item_text.split()[0])

        # 从数据结构中删除
        if frame_idx in self.polygons and poly_id in self.polygons[frame_idx]:
            del self.polygons[frame_idx][poly_id]

            del self.measurements[frame_idx][poly_id]

        # 如果多边形颜色字典中有，也删除
        if poly_id in self.polygon_colors:
            del self.polygon_colors[poly_id]

        self.display_current_frame()
        self.status_var.set(f"已删除多边形 #{poly_id}")

    def copy_to_next_frame(self):
        """将当前帧的多边形复制到下一帧"""
        frame_idx = self.current_frame

        if frame_idx not in self.polygons or not self.polygons[frame_idx]:
            messagebox.showwarning("警告", "当前帧没有多边形可复制")
            return

        if frame_idx >= self.total_frames - 1:
            messagebox.showwarning("警告", "已经是最后一帧")
            return

        next_frame = frame_idx + 1

        # 复制多边形
        if next_frame not in self.polygons:
            self.polygons[next_frame] = {}

        for poly_id, points in self.polygons[frame_idx].items():
            # 创建新的多边形ID
            new_poly_id = self.next_polygon_id
            self.next_polygon_id += 1

            # 复制点
            self.polygons[next_frame][new_poly_id] = points.copy()

            # 复制颜色
            if poly_id in self.polygon_colors:
                self.polygon_colors[new_poly_id] = self.polygon_colors[poly_id]

            # 重新计算面积
            area = self.calculate_polygon_area(points)
            if next_frame not in self.measurements:
                self.measurements[next_frame] = {}
            self.measurements[next_frame][new_poly_id] = area

        self.status_var.set(f"已将多边形复制到帧 #{next_frame + 1}")

    def zoom(self, factor):
        """缩放图像"""
        self.zoom_factor *= factor
        self.display_current_frame()

    def reset_view(self):
        """重置视图"""
        self.zoom_factor = 1.0
        self.pan_offset = [0, 0]
        self.display_current_frame()

    def on_canvas_scroll(self, event):
        """鼠标滚轮缩放"""
        if event.inaxes == self.ax:
            if event.button == 'up':
                self.zoom(1.2)
            elif event.button == 'down':
                self.zoom(0.8)

    def on_canvas_pan_start(self, event):
        """开始平移"""
        if event.inaxes == self.ax and event.button == 2:  # 中键
            self.pan_start = (event.xdata, event.ydata)

    def on_canvas_pan(self, event):
        """平移"""
        if self.pan_start and event.inaxes == self.ax:
            dx = event.xdata - self.pan_start[0]
            dy = event.ydata - self.pan_start[1]

            self.pan_offset[0] -= dx * self.zoom_factor
            self.pan_offset[1] -= dy * self.zoom_factor

            self.pan_start = (event.xdata, event.ydata)
            self.display_current_frame()

    def on_canvas_pan_end(self, event):
        """结束平移"""
        self.pan_start = None

    def export_data(self):
        """导出测量数据"""
        if not self.measurements:
            messagebox.showwarning("警告", "没有测量数据可导出")
            return

        # 选择保存位置
        file_path = filedialog.asksaveasfilename(
            title="导出数据",
            defaultextension=".csv",
            filetypes=[("CSV文件", "*.csv"), ("Excel文件", "*.xlsx"), ("所有文件", "*.*")]
        )

        if not file_path:
            return

        try:
            # 准备数据
            data = []
            for frame_idx in sorted(self.measurements.keys()):
                for poly_id in sorted(self.measurements[frame_idx].keys()):
                    area = self.measurements[frame_idx][poly_id]
                    points = self.polygons.get(frame_idx, {}).get(poly_id, [])

                    data.append({
                        '帧号': frame_idx + 1,
                        '多边形ID': poly_id,
                        '面积(像素²)': area,
                        '顶点数': len(points),
                        '颜色': self.polygon_colors.get(poly_id, 'N/A')
                    })

            # 创建DataFrame
            df = pd.DataFrame(data)

            # 保存文件
            if file_path.endswith('.xlsx'):
                df.to_excel(file_path, index=False)
            else:
                df.to_csv(file_path, index=False, encoding='utf-8-sig')

            # 同时导出详细的多边形坐标
            coord_file = file_path.replace('.csv', '_coordinates.csv').replace('.xlsx', '_coordinates.csv')
            coord_data = []
            for frame_idx in sorted(self.polygons.keys()):
                for poly_id in sorted(self.polygons[frame_idx].keys()):
                    points = self.polygons[frame_idx][poly_id]
                    for i, (x, y) in enumerate(points):
                        coord_data.append({
                            '帧号': frame_idx + 1,
                            '多边形ID': poly_id,
                            '顶点序号': i + 1,
                            'X坐标': x,
                            'Y坐标': y
                        })

            if coord_data:
                coord_df = pd.DataFrame(coord_data)
                coord_df.to_csv(coord_file, index=False, encoding='utf-8-sig')

            self.status_var.set(f"数据已导出到: {file_path}")
            messagebox.showinfo("成功",
                                f"数据已成功导出！\n\n主文件: {Path(file_path).name}\n坐标文件: {Path(coord_file).name if coord_data else '无'}")

        except Exception as e:
            messagebox.showerror("错误", f"导出数据失败: {str(e)}")

    def export_chart(self):
        """导出面积变化图表"""
        if not self.measurements:
            messagebox.showwarning("警告", "没有测量数据可导出图表")
            return

        # 选择保存位置
        file_path = filedialog.asksaveasfilename(
            title="导出图表",
            defaultextension=".png",
            filetypes=[("PNG图像", "*.png"), ("PDF文件", "*.pdf"), ("所有文件", "*.*")]
        )

        if not file_path:
            return

        try:
            # 创建图表
            fig, axes = plt.subplots(2, 2, figsize=(12, 10))

            # 1. 每个多边形的面积变化
            ax1 = axes[0, 0]
            polygon_data = {}

            for frame_idx in sorted(self.measurements.keys()):
                for poly_id, area in self.measurements[frame_idx].items():
                    if poly_id not in polygon_data:
                        polygon_data[poly_id] = {'frames': [], 'areas': []}
                    polygon_data[poly_id]['frames'].append(frame_idx + 1)
                    polygon_data[poly_id]['areas'].append(area)

            for poly_id, data in polygon_data.items():
                ax1.plot(data['frames'], data['areas'], 'o-', label=f'多边形 {poly_id}',
                         color=self.polygon_colors.get(poly_id, 'blue'))

            ax1.set_xlabel('帧号')
            ax1.set_ylabel('面积 (像素²)')
            ax1.set_title('每个多边形的面积变化')
            ax1.legend(loc='best', fontsize=8)
            ax1.grid(True, alpha=0.3)

            # 2. 每帧的总面积
            ax2 = axes[0, 1]
            frame_numbers = []
            total_areas = []

            for frame_idx in sorted(self.measurements.keys()):
                total_area = sum(self.measurements[frame_idx].values())
                frame_numbers.append(frame_idx + 1)
                total_areas.append(total_area)

            ax2.bar(frame_numbers, total_areas, color='skyblue', alpha=0.7)
            ax2.set_xlabel('帧号')
            ax2.set_ylabel('总面积 (像素²)')
            ax2.set_title('每帧的总面积')
            ax2.grid(True, alpha=0.3, axis='y')

            # 3. 每帧的多边形数量
            ax3 = axes[1, 0]
            frame_numbers = []
            polygon_counts = []

            for frame_idx in sorted(self.polygons.keys()):
                frame_numbers.append(frame_idx + 1)
                polygon_counts.append(len(self.polygons[frame_idx]))

            ax3.plot(frame_numbers, polygon_counts, 's-', color='green', linewidth=2, markersize=6)
            ax3.set_xlabel('帧号')
            ax3.set_ylabel('多边形数量')
            ax3.set_title('每帧的多边形数量')
            ax3.grid(True, alpha=0.3)

            # 4. 面积分布直方图

            ax4.set_xlabel('面积 (像素²)')
            ax4.set_ylabel('频率')
            ax4.set_title('面积分布直方图')
            ax4.grid(True, alpha=0.3, axis='y')

            # 调整布局
            plt.tight_layout()

            # 保存图表
            plt.savefig(file_path, dpi=300, bbox_inches='tight')
            plt.close(fig)

            self.status_var.set(f"图表已导出到: {file_path}")
            messagebox.showinfo("成功", f"图表已成功导出到:\n{file_path}")

        except Exception as e:
            messagebox.showerror("错误", f"导出图表失败: {str(e)}")

    def save_project(self):
        """保存项目"""
        if self.tiff_stack is None:
            messagebox.showwarning("警告", "没有加载图像，无法保存项目")
            return

        # 选择保存位置
        file_path = filedialog.asksaveasfilename(
            title="保存项目",
            defaultextension=".json",
            filetypes=[("JSON项目文件", "*.json"), ("所有文件", "*.*")]
        )

        if not file_path:
            return

        try:
            # 准备项目数据
            project_data = {
                'file_path': str(self.file_path),
                'total_frames': self.total_frames,
                'current_frame': self.current_frame,
                'next_polygon_id': self.next_polygon_id,
                'polygons': {},
                'polygon_colors': self.polygon_colors,
                'measurements': {}
            }

            # 保存多边形数据（转换为列表格式）
            for frame_idx, polygons in self.polygons.items():
                project_data['polygons'][str(frame_idx)] = {}
                for poly_id, points in polygons.items():
                    project_data['polygons'][str(frame_idx)][str(poly_id)] = points

            # 保存测量数据
            for frame_idx, measurements in self.measurements.items():
                project_data['measurements'][str(frame_idx)] = measurements

            # 保存到文件
            with open(file_path, 'w', encoding='utf-8') as f:
                json.dump(project_data, f, indent=2, ensure_ascii=False)

            self.status_var.set(f"项目已保存到: {file_path}")
            messagebox.showinfo("成功", f"项目已成功保存到:\n{file_path}")

        except Exception as e:
            messagebox.showerror("错误", f"保存项目失败: {str(e)}")

    def load_project(self):
        """加载项目"""
        file_path = filedialog.askopenfilename(
            title="加载项目",
            filetypes=[("JSON项目文件", "*.json"), ("所有文件", "*.*")]
        )

        if not file_path:
            return

        try:
            # 读取项目文件
            with open(file_path, 'r', encoding='utf-8') as f:
                project_data = json.load(f)

            # 加载TIFF文件
            tiff_path = project_data['file_path']
            if Path(tiff_path).exists():
                self.load_tiff_file(tiff_path)
            else:
                # 如果原始文件不存在用户选择
                messagebox.showwarning("文件不存在", f"原始TIFF文件不存在:\n{tiff_path}\n\n请选择新的TIFF文件。")
                self.open_tiff_file()
                if self.tiff_stack is None:
                    return

            # 恢复项目状态
            self.current_frame = project_data.get('current_frame', 0)
            self.next_polygon_id = project_data.get('next_polygon_id', 1)
            self.polygon_colors = project_data.get('polygon_colors', {})

            # 恢复多边形数据
            self.polygons = {}
            polygons_data = project_data.get('polygons', {})
            for frame_str, polygons in polygons_data.items():
                frame_idx = int(frame_str)
                self.polygons[frame_idx] = {}
                for poly_str, points in polygons.items():
                    poly_id = int(poly_str)
                    self.polygons[frame_idx][poly_id] = points

            # 恢复测量数据
            self.measurements = {}
            measurements_data = project_data.get('measurements', {})
            for frame_str, measurements in measurements_data.items():
                frame_idx = int(frame_str)
                self.measurements[frame_idx] = measurements

            # 更新UI
            self.frame_var.set(str(self.current_frame + 1))
            self.display_current_frame()

            self.status_var.set(f"项目已加载: {Path(file_path).name}")
            messagebox.showinfo("成功", f"项目已成功加载！\n\n已恢复 {len(self.polygons)} 帧的多边形数据。")

        except Exception as e:
            messagebox.showerror("错误", f"加载项目失败: {str(e)}")
